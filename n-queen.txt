def is_safe(board, row, col, n):

    # Check upper column
    for i in range(row):
        if board[i][col] == 1:
            return False

    # Check upper left diagonal
    i = row - 1
    j = col - 1

    while i >= 0 and j >= 0:
        if board[i][j] == 1:
            return False
        i -= 1
        j -= 1

    # Check upper right diagonal
    i = row - 1
    j = col + 1

    while i >= 0 and j < n:
        if board[i][j] == 1:
            return False
        i -= 1
        j += 1

    return True


def solve_nqueen(board, row, n):

    # All queens are placed
    if row == n:
        return True

    # Try each column
    for col in range(n):

        if is_safe(board, row, col, n):

            # Place queen
            board[row][col] = 1

            # Recursively place next queen
            if solve_nqueen(board, row + 1, n):
                return True

            # Backtracking: remove queen
            board[row][col] = 0

    return False


n = int(input("Enter number of queens: "))

board = [[0 for i in range(n)] for j in range(n)]

if solve_nqueen(board, 0, n):

    print("Solution:")
    for row in board:
        print(row)

else:

    print("No solution exists") 




Time	O(N!)	We try different column placements for each row, with backtracking.
Space	O(N²)	The chessboard requires N × N space.
